@bot.command()
async def say(ctx, *, arg=None):
    if arg is not None:
        message = ctx.message
        try:
            await message.delete()
        except:
            pass
        newarg = "".join(arg.split())
        args = list(newarg)
        if len(args) > 1:
            if '>' in args and ':' in args and '<' not in args and args[args.index('>')+1] == ':':
                args[list(args).index('>')] = ('<')
                arg = "".join(args)
        await ctx.send(arg)
        
    else:
        await ctx.send("please specify what to say")
